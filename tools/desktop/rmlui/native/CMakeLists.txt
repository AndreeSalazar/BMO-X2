cmake_minimum_required(VERSION 3.20)
project(BmoRmlUiNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(BMO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(dep ZLIB BZIP2 PNG HARFBUZZ BROTLI)
  set(FT_DISABLE_${dep} ON CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${BMO_ROOT}/build/deps/freetype-2.14.3" freetype)
if(NOT TARGET Freetype::Freetype)
  add_library(Freetype::Freetype ALIAS freetype)
endif()
set(RMLUI_SAMPLES OFF CACHE BOOL "" FORCE)
set(RMLUI_LUA_BINDINGS OFF CACHE BOOL "" FORCE)
set(RMLUI_PRECOMPILED_HEADERS OFF CACHE BOOL "" FORCE)
set(RMLUI_COMPILER_OPTIONS OFF CACHE BOOL "" FORCE)
add_subdirectory("${BMO_ROOT}/build/deps/RmlUi-6.3" rmlui)
# RmlUi includes sstream here without using it. libc++ deliberately omits
# localized streams in this target. Patch a build copy, never the pinned source.
set(original "${BMO_ROOT}/build/deps/RmlUi-6.3/Source/Core/StringUtilities.cpp")
file(READ "${original}" text)
if(NOT text MATCHES "#include <sstream>" OR text MATCHES "std::[io]*stringstream")
  message(FATAL_ERROR "Review the pinned StringUtilities stream dependency")
endif()
string(REPLACE "#include <sstream>" "// Unused stream include omitted in the native profile." text "${text}")
string(REPLACE "../../Include/" "" text "${text}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/StringUtilities.cpp" CONTENT "${text}")
get_target_property(core_sources rmlui_core SOURCES)
list(FILTER core_sources EXCLUDE REGEX "(^|/)StringUtilities\\.cpp$")
set_property(TARGET rmlui_core PROPERTY SOURCES "${core_sources}")
target_sources(rmlui_core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/StringUtilities.cpp")
# Limit the atlas construction scratch allocation to 256*256*4 (256 KiB).
# Glyph resolution is unchanged; large sets may use more atlas pages.
set(font_original "${BMO_ROOT}/build/deps/RmlUi-6.3/Source/Core/FontEngineDefault/FontFaceLayer.cpp")
file(READ "${font_original}" font_text)
if(NOT font_text MATCHES "constexpr int max_texture_dimensions = 1024;")
  message(FATAL_ERROR "Review pinned FreeType atlas dimensions")
endif()
string(REPLACE "constexpr int max_texture_dimensions = 1024;" "constexpr int max_texture_dimensions = 256;" font_text "${font_text}")
string(REGEX MATCHALL "#include \"[^\"]+\"" local_includes "${font_text}")
foreach(include IN LISTS local_includes)
  string(REGEX REPLACE "#include \"([^\"]+)\"" "\\1" relative "${include}")
  get_filename_component(full "${BMO_ROOT}/build/deps/RmlUi-6.3/Source/Core/FontEngineDefault/${relative}" ABSOLUTE)
  string(REPLACE "${include}" "#include \"${full}\"" font_text "${font_text}")
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/FontFaceLayer.cpp" CONTENT "${font_text}")
get_target_property(core_sources rmlui_core SOURCES)
list(FILTER core_sources EXCLUDE REGEX "(^|/)FontFaceLayer\\.cpp$")
set_property(TARGET rmlui_core PROPERTY SOURCES "${core_sources}")
target_sources(rmlui_core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/FontFaceLayer.cpp")
